cmake_minimum_required(VERSION 3.20)
project(listingturbo_native LANGUAGES CXX)
add_library(listingturbo_native SHARED src/listingturbo_native.cpp)
target_compile_features(listingturbo_native PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(listingturbo_native PRIVATE /O2 /EHsc)
else()
    target_compile_options(listingturbo_native PRIVATE -O3 -fvisibility=hidden)
    target_link_libraries(listingturbo_native PRIVATE dl)
endif()
set_target_properties(listingturbo_native PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/bin LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/bin ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/build)
